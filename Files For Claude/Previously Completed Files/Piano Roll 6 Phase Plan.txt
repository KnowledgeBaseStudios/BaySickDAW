// A utility function Claude can use to snap a Y-coordinate pitch to the nearest scale degree
function snapPitchToScale(rawPitch, rootNote, scaleIntervals) {
  const octave = Math.floor(rawPitch / 12);
  const pitchClass = rawPitch % 12;
  const normalizedPitch = (pitchClass - rootNote + 12) % 12;


  // Find the closest interval in the current scale
  let closestInterval = scaleIntervals.reduce((prev, curr) => {
    return (Math.abs(curr - normalizedPitch) < Math.abs(prev - normalizedPitch) ? curr : prev);
  });


  return (octave * 12) + rootNote + closestInterval;
}


5. The "Tools" Menu (Advanced Algorithms)
The manual mentions: Generate chord progression (Alt+P), Articulate (Alt+L), Quantize (Alt+Q), Chop (Alt/Opt+U), Glue (Ctrl+G), Arpeggiate (Alt+A), Strum (Alt+S), Randomize (Alt+R). Each of these is a distinct mathematical mutation applied to the selected MIDINote array.
* Quantize: Loops through selected notes and moves startTime to the nearest grid line.
* Chop: Takes a single note and divides its duration by a specific divisor, replacing the original note with an array of smaller repeating notes.
* Strum: Sorts a chord (notes sharing the same startTime) by pitch, and applies a slight, cascading time delay to each note's startTime to simulate a guitar strum.
* Arpeggiate: Converts a chord block into a sequential, repeating monophonic pattern.
